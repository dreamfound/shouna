## 3. 数据结构与接口（类图）

> 图中字段与方法签名可直接照抄实现。`Instant` 在 domain 层使用；Entity 层一律 `Long`。

```mermaid
classDiagram
    %% ============ 枚举 ============
    class ItemStatus {
        <<enum>>
        IN_STORAGE
        TO_BE_PUT_BACK
        DISCARDED
        GIVEN_AWAY
        USED_UP
        +code: String
        +isActive: Boolean
    }
    class LocationType {
        <<enum>>
        ROOM
        CABINET
        BOX
        SLOT
        OTHER
    }
    class ImportStrategy {
        <<enum>>
        OVERWRITE
        MERGE
    }
    class DeleteMode {
        <<enum>>

        MIGRATE_ITEMS

    %% ============ Entity ============
    class ItemEntity {

        +normalizedName: String

        +createdAt: Long

    class LocationEntity {
        +id: String
        +parentId: String?

        +isBuiltIn: Boolean
        +note: String?
        +sortOrder: Int
        +path: String
        +depth: Int
        +createdAt: Long
    }
    class PlacementEntity {
        +itemId: String

    class CategoryEntity {

    %% ============ Domain Model ============
    class Item {
        +id: String
        +name: String
        +categoryId: String?
        +status: ItemStatus
        +quantity: Int
        +note: String?
        +createdAt: Instant
        +lastModifiedAt: Instant
        +lastConfirmedAt: Instant?
    }
    class Location {
        +id: String
        +parentId: String?
        +name: String
        +type: LocationType?
        +isTemporary: Boolean
        +isBuiltIn: Boolean
        +note: String?
        +sortOrder: Int
        +path: String
        +depth: Int
    }
    class Placement {
        +itemId: String
        +locationId: String
        +isPrimary: Boolean
        +quantity: Int?
        +note: String?
        +lastConfirmedAt: Instant?
    }
    class ItemDetail {
        +item: Item
        +placements: List~Placement~
        +primaryLocationPath: List~String~
        +categoryName: String?
        +tags: List~String~
        +aliases: List~String~
    }
    class ItemListRow {
        +itemId: String
        +name: String
        +status: ItemStatus
        +quantity: Int
        +primaryLocationId: String?
        +primaryLocationPath: String?
        +categoryName: String?
        +isTemporaryLocation: Boolean
        +lastConfirmedAt: Instant?
        +lastModifiedAt: Instant
    }
    class LocationTreeRow {
        +id: String
        +parentId: String?
        +name: String
        +isTemporary: Boolean
        +isBuiltIn: Boolean
        +depth: Int
        +sortOrder: Int
        +directItemCount: Int
        +subtreeItemCount: Int
    }
    class SearchDoc {
        +itemId: String
        +name: String
        +normalizedName: String
        +pinyinFull: String
        +pinyinInitials: String
        +aliasBlob: String
        +categoryNameNormalized: String?
        +locationNamesNormalized: List~String~
        +tagNamesNormalized: List~String~
        +status: ItemStatus
        +lastConfirmedAt: Instant?
        +createdAt: Instant
    }
    class SearchHit {
        +doc: SearchDoc
        +score: Int
        +matchType: MatchType
        +highlightRanges: List~IntRange~
    }
    class SearchFilters {
        +categoryIds: Set~String~
        +locationIds: Set~String~
        +statuses: Set~ItemStatus~
        +onlyIncomplete: Boolean
    }
    class MatchType {
        <<enum>>
        NAME_PREFIX
        NAME_SUBSTRING
        ALIAS
        PINYIN_INITIALS
        PINYIN_FULL
        TAG
        LOCATION
        CATEGORY
    }

    %% ============ Search Engine ============
    class SearchIndex {
        -docs: List~SearchDoc~
        +query(raw: String, filters: SearchFilters, includeInactive: Boolean, limit: Int): List~SearchHit~
        -score(doc, qNorm, qInitials): SearchHit?
    }
    class SearchScorer {
        +weight(type: MatchType): Int
        +compare(a: SearchHit, b: SearchHit): Int
    }

    %% ============ DAO ============

        +updateSubtreePath(oldPrefix: String, newPrefix: String, deltaDepth: Int)

    %% ============ Relationships ============
    ItemEntity "1" --> "0..1" CategoryEntity : categoryId (SET NULL)
    ItemEntity "1" --> "*" PlacementEntity : placements (CASCADE)
    LocationEntity "1" --> "*" PlacementEntity : placements (CASCADE)
    LocationEntity "0..1" --> "*" LocationEntity : parentId (RESTRICT)
    ItemEntity "1" --> "*" ItemAliasEntity : aliases (CASCADE)
    ItemEntity "1" --> "*" ItemTagCrossRef : itemTag
    TagEntity "1" --> "*" ItemTagCrossRef : itemTag
    LocationEntity "1" --> "0..1" RecentLocationEntity : recent (CASCADE)

    ItemEntity ..> Item : mapsTo
    LocationEntity ..> Location : mapsTo
    PlacementEntity ..> Placement : mapsTo
    Item ..> ItemStatus : uses
    ItemDetail *-- Item

    SearchIndex o-- SearchDoc

    SearchHit *-- SearchDoc
```
